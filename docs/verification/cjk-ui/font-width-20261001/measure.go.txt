package main

import (
	"encoding/json"
	"os"

	"github.com/wicanr2/kol-remake-cht/internal/font"
)

func must(err error) { if err != nil { panic(err) } }
type input struct { Checks []struct { Text string `json:"zh"` } `json:"checks"` }
func main() {
	f,err:=font.Load("assets/ui-font","zh-hant-16");must(err)
	raw,err:=os.ReadFile("workplace/font-space-20261001/text-fit.json");must(err)
	var in input;must(json.Unmarshal(raw,&in))
	widths:=make(map[string]int)
	for _,row:=range in.Checks { if row.Text!="" {w,_:=f.Measure(row.Text);widths[row.Text]=w} }
	for _,text:=range []string{" ","A A","A  A","A\u3000A"} {w,_:=f.Measure(text);widths[text]=w}
	b,err:=json.MarshalIndent(widths,"","  ");must(err)
	must(os.WriteFile("workplace/font-space-20261001/runtime-widths.json",append(b,'\n'),0644))
}
