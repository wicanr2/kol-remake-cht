type positionCityFrame struct {Case,Variant,PNG,SHA,StateSHA string;Text []string;Han int}

func(g *positionCityGallery)Draw(dst *ebiten.Image){

 g.j++;if g.j==4{g.j=0;g.i++;g.done=g.i==len(g.cases);fmt.Printf("position city%d/%d\n",g.i,len(g.cases))}
