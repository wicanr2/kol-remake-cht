type fullNPCScene struct {ID string `json:"id"`;Kind string `json:"kind"`;Archive string `json:"archive"`;Index int `json:"index"`;In *town.Inn `json:"-"`}
type fullNPCFrame struct {ID string `json:"id"`;Kind string `json:"kind"`;Archive string `json:"archive"`;Index int `json:"index"`;Variant string `json:"variant"`;PNG string `json:"png"`;SHA string `json:"sha256"`;StateSHA string `json:"inn_sha256"`;Text []string `json:"text"`;Native bool `json:"native_ui"`;Han int `json:"han_characters"`}
type fullNPCGallery struct {models [2]*game;cases []fullNPCScene;frames []fullNPCFrame;i,j int;err error;done bool;dir string;before map[string][]byte;text map[string][]string}
func(g *fullNPCGallery)Update()error{if g.err!=nil{return g.err};if g.done{return ebiten.Termination};return nil}
func(g *fullNPCGallery)Layout(int,int)(int,int){return 1920,1200}
func(g *fullNPCGallery)Draw(dst *ebiten.Image){
 if g.done||g.err!=nil{return};c:=g.cases[g.i];model:=g.models[g.j/2];theme:="ega";if g.j%2==1{theme="hd"};variant:="baseline-"+theme;if g.j>=2{variant="prototype-"+theme}
 model.scene=sceneBuilding;model.inn=c.In;state,_:=json.Marshal(c.In)
 if g.j==0{g.before[c.ID]=state}else if string(state)!=string(g.before[c.ID]){g.err=fmt.Errorf("initial state changed %s",c.ID);return}
 if g.err=model.setGraphics(theme);g.err!=nil{return};var words []string
 font.DrawObserver=func(_,s string){words=append(words,s)};g.err=model.render();font.DrawObserver=nil;if g.err!=nil{return}
 if g.j==0{g.text[c.ID]=words}else if !reflect.DeepEqual(g.text[c.ID],words){g.err=fmt.Errorf("render text changed %s/%s",c.ID,variant);return}
 if g.err=model.presentHD();g.err!=nil{return};after,_:=json.Marshal(c.In);if string(after)!=string(state){g.err=fmt.Errorf("render mutated Inn %s",c.ID);return}
 im:=model.outputCanvas().Image();b:=im.Bounds();pixels:=make([]byte,b.Dx()*b.Dy()*4);im.ReadPixels(pixels);rgba:=image.NewRGBA(image.Rect(0,0,b.Dx(),b.Dy()));copy(rgba.Pix,pixels)
 name:=c.ID+"-"+variant+".png";f,e:=os.OpenFile(filepath.Join(g.dir,name),os.O_CREATE|os.O_EXCL|os.O_WRONLY,0644);if e!=nil{g.err=e;return};encoder:=png.Encoder{CompressionLevel:png.BestSpeed};e=encoder.Encode(f,rgba);ce:=f.Close();if e==nil{e=ce};if e!=nil{g.err=e;return};raw,e:=os.ReadFile(filepath.Join(g.dir,name));if e!=nil{g.err=e;return}
 han:=0;for _,s:=range words{for _,r:=range s{if unicode.Is(unicode.Han,r){han++}}};if han==0{g.err=fmt.Errorf("no Chinese %s",c.ID);return}
 g.frames=append(g.frames,fullNPCFrame{ID:c.ID,Kind:c.Kind,Archive:c.Archive,Index:c.Index,Variant:variant,PNG:name,SHA:fmt.Sprintf("%x",sha256.Sum256(raw)),StateSHA:fmt.Sprintf("%x",sha256.Sum256(state)),Text:words,Native:model.nativeUIPage(),Han:han})
 if theme=="hd"{dst.DrawImage(im,nil)}else{var op ebiten.DrawImageOptions;op.GeoM.Scale(2,2);dst.DrawImage(im,&op)}
 g.j++;if g.j==4{g.j=0;g.i++;if g.i%10==0{fmt.Printf("full NPC gallery %d/%d scenes\n",g.i,len(g.cases))};g.done=g.i==len(g.cases)}
}
