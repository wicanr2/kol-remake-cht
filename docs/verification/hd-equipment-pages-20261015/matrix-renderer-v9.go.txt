type equipmentPartPrototype struct {
	File  string    `json:"file"`
	Rect  []int     `json:"rect"`
	World []float64 `json:"world"`
	Kind  string    `json:"kind"`
}
type equipmentPropPrototype struct {
	File       string `json:"file"`
	Rect       []int  `json:"rect"`
	Paint      []int  `json:"paint"`
	SourceSize []int  `json:"source_size"`
}
type equipmentPrototype struct {
	Parts          map[string]equipmentPartPrototype `json:"parts"`
	Props          map[string]equipmentPropPrototype
	HandPaint      map[string][]int
	ControlSources map[string]string `json:"control_sources"`
	Images         map[string]*ebiten.Image
	Sources        map[string]*ebiten.Image
	Base, Figure   *ebiten.Image
	GripsV7        map[string][]equipmentGripV7 `json:"grips"`
	MeshesV7       map[string]equipmentMeshV7   `json:"meshes_v7"`
	SourceHandsV7  map[string][]int             `json:"source_hands"`
	BaseOriginsV7  map[string]int               `json:"base_origins"`
	BaseWaistsV7   map[string]float64           `json:"base_waists"`
	HeadBandsV7    map[string][]int             `json:"head_bands"`
	WarpedV7       map[string]*ebiten.Image
	BodyShapesV7   map[string]*ebiten.Image
	BodyBoundsV7   map[int]image.Rectangle
}

var equipmentDraft *equipmentPrototype

func loadEquipmentPrototype(n *hdNativeRender) (*equipmentPrototype, error) {
	if equipmentDraft != nil {
		return equipmentDraft, nil
	}
	raw, err := os.ReadFile("/out/segment-prototype-v7.json")
	if err != nil {
		return nil, err
	}
	p := &equipmentPrototype{Images: map[string]*ebiten.Image{}}
	if err = json.Unmarshal(raw, p); err != nil {
		return nil, err
	}
	raw, err = os.ReadFile("/out/props-prototype.json")
	if err != nil {
		return nil, err
	}
	var props struct {
		Schema    string                            `json:"schema"`
		Props     map[string]equipmentPropPrototype `json:"props"`
		HandPaint map[string][]int                  `json:"hand_paint"`
	}
	if err = json.Unmarshal(raw, &props); err != nil {
		return nil, err
	}
	if props.Schema != "kol-hd-equipment-props-prototype/1" || len(props.Props) != 32 {
		return nil, fmt.Errorf("DRAFT武器盾牌清冊不符")
	}
	p.Props = props.Props
	p.HandPaint = props.HandPaint
	for _, archive := range []string{"WEAPPICS", "SHLDPICS"} {
		count := 28
		if archive == "SHLDPICS" {
			count = 4
		}
		for i := 0; i < count; i++ {
			part, ok := p.Props[fmt.Sprintf("%s:%d", archive, i)]
			if !ok || len(part.Rect) != 4 || len(part.Paint) != 4 || len(part.SourceSize) != 2 {
				return nil, fmt.Errorf("DRAFT武器盾牌圖號或矩形不符")
			}
			if _, ok := p.Images[part.File]; !ok {
				if p.Images[part.File], err = loadEquipmentDraftPNG(part.File); err != nil {
					return nil, err
				}
			}
			r := image.Rect(part.Rect[0], part.Rect[1], part.Rect[2], part.Rect[3])
			b := image.Rect(part.Paint[0], part.Paint[1], part.Paint[2], part.Paint[3])
			if r.Empty() || !r.In(p.Images[part.File].Bounds()) || b.Empty() || !b.In(image.Rect(0, 0, part.SourceSize[0], part.SourceSize[1])) {
				return nil, fmt.Errorf("DRAFT武器盾牌裁取越界")
			}
		}
	}
	for _, part := range p.Parts {
		if len(part.Rect) != 4 || len(part.World) != 4 {
			return nil, fmt.Errorf("DRAFT部件矩形不符")
		}
		if _, ok := p.Images[part.File]; ok {
			continue
		}
		if p.Images[part.File], err = loadEquipmentDraftPNG(part.File); err != nil {
			return nil, err
		}
	}
	for _, name := range []string{"town-back-painted.png", "outdoor-back-painted.png"} {
		if p.Images[name], err = loadEquipmentDraftPNG(name); err != nil {
			return nil, err
		}
	}
	p.Sources = map[string]*ebiten.Image{}
	p.WarpedV7 = map[string]*ebiten.Image{}
	p.BodyShapesV7 = map[string]*ebiten.Image{}
	p.BodyBoundsV7 = map[int]image.Rectangle{}
	if len(p.GripsV7) != 50 || len(p.BaseOriginsV7) != 6 {
		return nil, fmt.Errorf("DRAFT六種體型清冊不符")
	}
	for key, part := range p.Parts {
		p.Sources[key] = p.Images[part.File].SubImage(image.Rect(part.Rect[0], part.Rect[1], part.Rect[2], part.Rect[3])).(*ebiten.Image)
	}
	for key, part := range p.Props {
		p.Sources[key] = p.Images[part.File].SubImage(image.Rect(part.Rect[0], part.Rect[1], part.Rect[2], part.Rect[3])).(*ebiten.Image)
	}
	p.Base, p.Figure = ebiten.NewImage(1920, 1200), ebiten.NewImage(1920, 1200)
	eye := ebiten.NewImage(144, 120)
	points := [][2]float32{{12, 60}, {44, 34}, {100, 34}, {132, 60}, {100, 86}, {44, 86}, {12, 60}}
	for i := 1; i < len(points); i++ {
		vector.StrokeLine(eye, points[i-1][0], points[i-1][1], points[i][0], points[i][1], 5, hdGold, true)
	}
	vector.StrokeCircle(eye, 72, 60, 20, 5, hdGold, true)
	vector.DrawFilledCircle(eye, 72, 60, 7, hdGold, true)
	space := ebiten.NewImage(144, 120)
	vector.StrokeRect(space, 12, 35, 120, 56, 5, hdGold, true)
	vector.StrokeLine(space, 18, 99, 126, 99, 4, hdGold, true)
	for _, ch := range "空白" {
		if _, ok := n.h.style.glyphs[string(ch)]; !ok {
			return nil, fmt.Errorf("DRAFT空白鍵缺字：%c", ch)
		}
	}
	n.h.style.label(space, "空白", image.Rect(20, 46, 124, 82), 26, hdGold, true)
	n.h.pages.icons[p.ControlSources["12"]] = eye
	n.h.pages.icons[p.ControlSources["4"]] = space
	equipmentDraft = p
	return p, nil
}
func loadEquipmentDraftPNG(name string) (*ebiten.Image, error) {
	f, err := os.Open(filepath.Join("/out/generated", name))
	if err != nil {
		return nil, err
	}
	defer f.Close()
	im, err := png.Decode(f)
	if err != nil {
		return nil, err
	}
	return ebiten.NewImageFromImage(im), nil
}
func (p *equipmentPrototype) part(n *hdNativeRender, dst *ebiten.Image, key string) {
	a := p.Parts[key]
	src := p.Sources[key]
	n.h.drawHD(dst, src, equipmentRectV7(a.World))
}
func drawEquipmentBackdropPrototype(dst *ebiten.Image, wild bool) (bool, error) {
	n := activeHDUI
	if n == nil || dst != n.dst {
		return false, nil
	}
	p, err := loadEquipmentPrototype(n)
	if err != nil {
		return true, err
	}
	name := "town-back-painted.png"
	if wild {
		name = "outdoor-back-painted.png"
	}
	n.h.drawHD(dst, p.Images[name], image.Rect(96, 18, 1824, 948))
	return true, nil
}
func drawEquipmentPrototype(dst *ebiten.Image, sprites Sprites, pieces []portrait.Piece, scale int) (bool, error) {
	n := activeHDUI
	if n == nil || dst != n.dst {
		return false, nil
	}
	p, err := loadEquipmentPrototype(n)
	if err != nil {
		return true, err
	}
	for first := 0; first < len(pieces); {
		last := first + 1
		for last < len(pieces) && pieces[last].Archive != "PERPICS" {
			last++
		}
		group := pieces[first:last]
		if group[0].Archive == "PERPICS" && group[0].Index > 0 && group[0].Index < 6 {
			if err := p.figureV7(n, dst, sprites, group); err != nil {
				return true, err
			}
			first = last
			continue
		}
		if group[0].Archive != "PERPICS" || group[0].Index != 0 {
			for _, q := range group {
				pic, e := pieceImage(sprites, q)
				if e != nil {
					return true, e
				}
				drawScaled(dst, pic, q.X*8*scale, q.Y*scale, scale)
			}
			first = last
			continue
		}
		indices := map[string]int{"HEADPICS": -1, "TORPICS": -1, "LEGSPICS": -1}
		for _, q := range group {
			if _, ok := indices[q.Archive]; ok {
				if q.Index < 0 || q.Index >= 9 {
					return true, fmt.Errorf("DRAFT體型0裝備圖號越界")
				}
				indices[q.Archive] = q.Index
			}
		}
		head, torso, legs := indices["HEADPICS"], indices["TORPICS"], indices["LEGSPICS"]
		p.Base.Clear()
		p.part(n, p.Base, "PERPICS:0")
		p.Figure.Clear()
		legTop := 924
		if legs >= 0 {
			legTop = int(math.Round(p.Parts[fmt.Sprintf("LEGSPICS:%d", legs)].World[1] * 6))
		}
		for y := 0; y < 1200; {
			visible := func(row int) bool {
				upper := 402
				if head >= 0 {
					upper = 390
				}
				return head < 0 && row < 402 || torso < 0 && row >= upper && row < legTop || legs < 0 && row >= 564
			}
			if !visible(y) {
				y++
				continue
			}
			end := y + 1
			for end < 1200 && visible(end) {
				end++
			}
			clip := image.Rect(0, y, 1920, end)
			p.Figure.SubImage(clip).(*ebiten.Image).DrawImage(p.Base, nil)
			y = end
		}
		if torso < 0 && legs >= 0 {
			for _, clip := range []image.Rectangle{image.Rect(756, 600, 852, 684), image.Rect(1068, 600, 1164, 684)} {
				p.Figure.SubImage(clip).(*ebiten.Image).DrawImage(p.Base, nil)
			}
		}
		for _, archive := range []string{"LEGSPICS", "TORPICS", "HEADPICS"} {
			if index := indices[archive]; index >= 0 {
				p.part(n, p.Figure, fmt.Sprintf("%s:%d", archive, index))
			}
		}
		dx := (group[0].X - 15) * 48
		var op ebiten.DrawImageOptions
		op.GeoM.Translate(float64(dx), 0)
		dst.DrawImage(p.Figure, &op)
		for _, q := range group[1:] {
			if _, ok := indices[q.Archive]; ok {
				continue
			}
			if q.Archive == "SHLDPICS" || q.Archive == "WEAPPICS" {
				key := fmt.Sprintf("%s:%d", q.Archive, q.Index)
				a, ok := p.Props[key]
				if !ok || q.W*8 != a.SourceSize[0] || q.H != a.SourceSize[1] {
					return true, fmt.Errorf("DRAFT武器盾牌來源尺寸不符：%s", key)
				}
				n.h.drawHD(dst, p.Sources[key], image.Rect((q.X*8+a.Paint[0])*6, (q.Y+a.Paint[1])*6, (q.X*8+a.Paint[2])*6, (q.Y+a.Paint[3])*6))
				continue
			}
			if q.Archive == "HANDPICS" {
				x, y := (q.X-(group[0].X-15))*48, q.Y*6
				b := p.HandPaint[fmt.Sprint(q.Index)]
				if len(b) != 4 || b[0] < 0 || b[1] < 0 || b[2] > q.W*8 || b[3] > q.H || b[2] <= b[0] || b[3] <= b[1] {
					return true, fmt.Errorf("DRAFT手部著色範圍不符")
				}
				r := image.Rect(x+b[0]*6, y+b[1]*6, x+b[2]*6, y+b[3]*6).Intersect(p.Figure.Bounds())
				if r.Empty() {
					return true, fmt.Errorf("DRAFT手部越界")
				}
				var hand ebiten.DrawImageOptions
				hand.GeoM.Translate(float64(r.Min.X+dx), float64(r.Min.Y))
				dst.DrawImage(p.Figure.SubImage(r).(*ebiten.Image), &hand)
				continue
			}
			if q.Archive == portrait.ArchiveDefaultShield {
				if err := p.shieldFaceV8(n, dst, sprites, group, 0, torso, dx, q); err != nil {
					return true, err
				}
				continue
			}
			pic, e := pieceImage(sprites, q)
			if e != nil {
				return true, e
			}
			drawScaled(dst, pic, q.X*8*scale, q.Y*scale, scale)
		}
		first = last
	}
	return true, nil
}

// V7 remains an isolated DRAFT. The mesh changes presentation only; fists use
// a uniform scale at the outer edges, and all original typed records remain.
type equipmentGripV7 struct {
	Rect       []int `json:"rect"`
	NativeRect []int `json:"native_rect"`
	HandIndex  int   `json:"hand_index"`
}
type equipmentMeshV7 struct {
	NativeWaist int     `json:"native_waist"`
	WaistWorld  float64 `json:"waist_world"`
}

func equipmentRectV7(a []float64) image.Rectangle {
	return image.Rect(int(math.Round(a[0]*6)), int(math.Round(a[1]*6)), int(math.Round(a[2]*6)), int(math.Round(a[3]*6)))
}
func (p *equipmentPrototype) torsoV7(n *hdNativeRender, key string) *ebiten.Image {
	if pic := p.WarpedV7[key]; pic != nil {
		return pic
	}
	part := p.Parts[key]
	src := p.Sources[key]
	b := src.Bounds()
	// Cache both input and output outside the automatic texture atlas. Their
	// normalized origins keep the same mesh independent of atlas placement.
	pixels := make([]byte, b.Dx()*b.Dy()*4)
	src.ReadPixels(pixels)
	stable := ebiten.NewImageWithOptions(image.Rect(0, 0, b.Dx(), b.Dy()), &ebiten.NewImageOptions{Unmanaged: true})
	stable.WritePixels(pixels)
	p.WarpedV7[key+":native"] = stable
	src, b = stable, stable.Bounds()
	r := equipmentRectV7(part.World)
	m := p.MeshesV7[key]
	sx := float64(r.Dx()) / float64(b.Dx())
	height := int(math.Ceil(math.Max(float64(r.Dy()), float64(b.Dy())*sx)))
	pic := ebiten.NewImageWithOptions(image.Rect(0, 0, r.Dx(), height), &ebiten.NewImageOptions{Unmanaged: true})
	xs := []float64{0, .25, .4, .6, .75, 1}
	ys := []float64{0, float64(b.Dy()) * .2, float64(b.Dy()) * .4, float64(m.NativeWaist), float64(b.Dy()) * .8, float64(b.Dy())}
	sort.Float64s(ys)
	verts := make([]ebiten.Vertex, 0, len(xs)*len(ys))
	for _, y := range ys {
		for _, x := range xs {
			weight := 1.0
			if x <= .25 || x >= .75 {
				weight = 0
			} else if x < .4 {
				weight = (x - .25) / .15
			} else if x > .6 {
				weight = (.75 - x) / .15
			}
			center := 0.0
			waist := m.WaistWorld*6 - float64(r.Min.Y)
			if y <= float64(m.NativeWaist) {
				center = y / float64(m.NativeWaist) * waist
			} else {
				center = waist + (y-float64(m.NativeWaist))/float64(b.Dy()-m.NativeWaist)*(float64(r.Dy())-waist)
			}
			dy := y*sx*(1-weight) + center*weight
			verts = append(verts, ebiten.Vertex{DstX: float32(x * float64(r.Dx())), DstY: float32(dy), SrcX: float32(float64(b.Min.X) + x*float64(b.Dx())), SrcY: float32(float64(b.Min.Y) + y), ColorR: 1, ColorG: 1, ColorB: 1, ColorA: 1})
		}
	}
	indices := []uint16{}
	for y := 0; y < len(ys)-1; y++ {
		for x := 0; x < len(xs)-1; x++ {
			a := uint16(y*len(xs) + x)
			b := a + 1
			c := a + uint16(len(xs))
			d := c + 1
			indices = append(indices, a, b, c, b, d, c)
		}
	}
	pic.DrawTriangles(verts, indices, src, &ebiten.DrawTrianglesOptions{Filter: ebiten.FilterLinear, Address: ebiten.AddressClampToZero})
	p.WarpedV7[key] = pic
	return pic
}
func (p *equipmentPrototype) bodyV7(n *hdNativeRender, build, mask int) (*ebiten.Image, image.Rectangle) {
	key := fmt.Sprintf("%d:%d", build, mask)
	if pic := p.BodyShapesV7[key]; pic != nil {
		return pic, p.BodyBoundsV7[build]
	}
	// This runs in Draw, after the graphics context is ready.
	p.Base.Clear()
	p.part(n, p.Base, fmt.Sprintf("PERPICS:%d", build))
	pixels := make([]byte, 1920*1200*4)
	p.Base.ReadPixels(pixels)
	r := equipmentRectV7(p.Parts[fmt.Sprintf("PERPICS:%d", build)].World)
	waist := int(math.Round(p.BaseWaistsV7[fmt.Sprint(build)] * 6))
	head := p.HeadBandsV7[fmt.Sprint(build)]
	arms := make(map[int][]image.Rectangle)
	limit := 0
	for side := 0; side < 2; side++ {
		if v := p.SourceHandsV7[fmt.Sprint(build*2+side)][3]*6 + 60; v > limit {
			limit = v
		}
	}
	for y := waist; y < min(limit, r.Max.Y); y++ {
		runs := []image.Rectangle{}
		start := -1
		for x := r.Min.X; x < r.Max.X; x++ {
			on := pixels[(y*1920+x)*4+3] >= 8
			if on && start < 0 {
				start = x
			}
			if !on && start >= 0 {
				if x-start >= 3 {
					runs = append(runs, image.Rect(start, y, x, y+1))
				}
				start = -1
			}
		}
		if start >= 0 && r.Max.X-start >= 3 {
			runs = append(runs, image.Rect(start, y, r.Max.X, y+1))
		}
		if len(runs) >= 3 {
			arms[y] = []image.Rectangle{runs[0], runs[len(runs)-1]}
		}
	}
	rgba := make([]byte, r.Dx()*r.Dy()*4)
	for y := r.Min.Y; y < r.Max.Y; y++ {
		for x := r.Min.X; x < r.Max.X; x++ {
			isHead := y < head[0]*6 && x >= head[1]*6 && x < head[2]*6
			keep := false
			if y < waist {
				keep = isHead && mask&1 == 0 || !isHead && mask&2 == 0
			} else {
				isArm := false
				for _, a := range arms[y] {
					if x >= a.Min.X && x < a.Max.X {
						isArm = true
						break
					}
				}
				keep = isArm && mask&2 == 0 || !isArm && mask&4 == 0
			}
			if keep {
				dst := ((y-r.Min.Y)*r.Dx() + x - r.Min.X) * 4
				src := (y*1920 + x) * 4
				copy(rgba[dst:dst+4], pixels[src:src+4])
			}
		}
	}
	pic := ebiten.NewImage(r.Dx(), r.Dy())
	pic.WritePixels(rgba)
	p.BodyShapesV7[key] = pic
	p.BodyBoundsV7[build] = r
	return pic, r
}
func rectFromGripV7(g equipmentGripV7) image.Rectangle {
	return image.Rect(g.Rect[0], g.Rect[1], g.Rect[2], g.Rect[3])
}
func centerV7(r image.Rectangle) (float64, float64) {
	return float64(r.Min.X+r.Max.X) / 2, float64(r.Min.Y+r.Max.Y) / 2
}
func (p *equipmentPrototype) propRectV7(group []portrait.Piece, body, torso, dx int, q portrait.Piece, a equipmentPropPrototype) image.Rectangle {
	r := image.Rect((q.X*8+a.Paint[0])*6, (q.Y+a.Paint[1])*6, (q.X*8+a.Paint[2])*6, (q.Y+a.Paint[3])*6)
	hands := []int{}
	for _, s := range group {
		if s.Archive == "HANDPICS" {
			hands = append(hands, s.Index)
		}
	}
	side := 0
	if q.Archive == "SHLDPICS" {
		side = 1
	} else if len(hands) == 0 {
		return r
	} else {
		side = hands[0] - body*2
	}
	source := func(side int) (float64, float64) {
		b := p.SourceHandsV7[fmt.Sprint(body*2+side)]
		return float64(b[0]+b[2])*3 + float64(dx), float64(b[1]+b[3]) * 3
	}
	target := func(side int) (float64, float64) {
		x, y := centerV7(rectFromGripV7(p.GripsV7[fmt.Sprintf("%d:%d", body, torso)][side]))
		return x + float64(dx), y
	}
	sx, sy := source(side)
	tx, ty := target(side)
	s := 1.0
	if q.Archive == "WEAPPICS" && len(hands) == 2 {
		sx, sy = source(0)
		x1, y1 := source(1)
		tx, ty = target(0)
		u1, v1 := target(1)
		s = (u1 - tx) / (x1 - sx)
		sy = (sy + y1) / 2
		ty = (ty + v1) / 2
	}
	ox, oy := tx-s*sx, ty-s*sy
	return image.Rect(int(math.Round(float64(r.Min.X)*s+ox)), int(math.Round(float64(r.Min.Y)*s+oy)), int(math.Round(float64(r.Max.X)*s+ox)), int(math.Round(float64(r.Max.Y)*s+oy)))
}
func (p *equipmentPrototype) figureV7(n *hdNativeRender, dst *ebiten.Image, sprites Sprites, group []portrait.Piece) error {
	body := group[0].Index
	indices := map[string]int{"HEADPICS": -1, "TORPICS": -1, "LEGSPICS": -1}
	for _, q := range group {
		if _, ok := indices[q.Archive]; ok {
			if q.Index < body*9 || q.Index >= body*9+9 {
				return fmt.Errorf("DRAFT體型與裝備圖號不符")
			}
			indices[q.Archive] = q.Index
		}
	}
	mask := 0
	if indices["HEADPICS"] >= 0 {
		mask |= 1
	}
	if indices["TORPICS"] >= 0 {
		mask |= 2
	}
	if leg := indices["LEGSPICS"]; leg >= 0 && p.Parts[fmt.Sprintf("LEGSPICS:%d", leg)].Kind == "" {
		mask |= 4
	}
	pic, r := p.bodyV7(n, body, mask)
	p.Figure.Clear()
	if body == 3 {
		p.part(n, p.Figure, "WINGS:3")
	}
	var base ebiten.DrawImageOptions
	base.GeoM.Translate(float64(r.Min.X), float64(r.Min.Y))
	p.Figure.DrawImage(pic, &base)
	for _, archive := range []string{"LEGSPICS", "TORPICS", "HEADPICS"} {
		if i := indices[archive]; i >= 0 {
			key := fmt.Sprintf("%s:%d", archive, i)
			if _, ok := p.MeshesV7[key]; ok {
				r := equipmentRectV7(p.Parts[key].World)
				var op ebiten.DrawImageOptions
				op.GeoM.Translate(float64(r.Min.X), float64(r.Min.Y))
				p.Figure.DrawImage(p.torsoV7(n, key), &op)
			} else {
				p.part(n, p.Figure, key)
			}
		}
	}
	dx := (group[0].X - p.BaseOriginsV7[fmt.Sprint(body)]) * 48
	var op ebiten.DrawImageOptions
	op.GeoM.Translate(float64(dx), 0)
	dst.DrawImage(p.Figure, &op)
	for _, q := range group[1:] {
		if _, ok := indices[q.Archive]; ok {
			continue
		}
		if q.Archive == "WEAPPICS" || q.Archive == "SHLDPICS" {
			key := fmt.Sprintf("%s:%d", q.Archive, q.Index)
			a, ok := p.Props[key]
			if !ok || q.W*8 != a.SourceSize[0] || q.H != a.SourceSize[1] {
				return fmt.Errorf("DRAFT武器來源不符")
			}
			r := p.propRectV7(group, body, indices["TORPICS"], dx, q, a)
			if r.Empty() {
				return fmt.Errorf("DRAFT持握矩形不符")
			}
			n.h.drawHD(dst, p.Sources[key], r)
			continue
		}
		if q.Archive == "HANDPICS" {
			side := q.Index - body*2
			g := p.GripsV7[fmt.Sprintf("%d:%d", body, indices["TORPICS"])]
			if side < 0 || side >= len(g) {
				return fmt.Errorf("DRAFT手部圖號不符")
			}
			r := rectFromGripV7(g[side]).Intersect(p.Figure.Bounds())
			if r.Empty() {
				return fmt.Errorf("DRAFT手部矩形不符")
			}
			var hand ebiten.DrawImageOptions
			hand.GeoM.Translate(float64(r.Min.X+dx), float64(r.Min.Y))
			dst.DrawImage(p.Figure.SubImage(r).(*ebiten.Image), &hand)
			continue
		}
		if q.Archive == portrait.ArchiveDefaultShield {
			if err := p.shieldFaceV8(n, dst, sprites, group, body, indices["TORPICS"], dx, q); err != nil {
				return err
			}
			continue
		}
		raw, err := pieceImage(sprites, q)
		if err != nil {
			return err
		}
		drawScaled(dst, raw, q.X*48, q.Y*6, 6)
	}
	return nil
}

// shieldFaceV8 keeps the editable bitmap's pixels and follows its own template.
// Only presentation coordinates change; Figure, Piece and saved bitmap stay intact.
func (p *equipmentPrototype) shieldFaceRectV8(group []portrait.Piece, body, torso, dx int, face portrait.Piece) (image.Rectangle, image.Rectangle, error) {
	var parent *portrait.Piece
	for i := range group {
		q := &group[i]
		if q.Archive == "SHLDPICS" && q.Index == portrait.PersonalShieldStyle {
			parent = q
			break
		}
	}
	if parent == nil {
		return image.Rectangle{}, image.Rectangle{}, fmt.Errorf("DRAFT個人盾牌缺少模板")
	}
	a, ok := p.Props[fmt.Sprintf("SHLDPICS:%d", parent.Index)]
	if !ok {
		return image.Rectangle{}, image.Rectangle{}, fmt.Errorf("DRAFT個人盾牌模板不符")
	}
	original := image.Rect((parent.X*8+a.Paint[0])*6, (parent.Y+a.Paint[1])*6, (parent.X*8+a.Paint[2])*6, (parent.Y+a.Paint[3])*6)
	target := original
	if body != 0 {
		target = p.propRectV7(group, body, torso, dx, *parent, a)
	}
	if original.Empty() || target.Empty() {
		return image.Rectangle{}, image.Rectangle{}, fmt.Errorf("DRAFT個人盾牌矩形不符")
	}
	sx, sy := float64(target.Dx())/float64(original.Dx()), float64(target.Dy())/float64(original.Dy())
	x := func(v int) int { return int(math.Round(float64(target.Min.X) + float64(v-original.Min.X)*sx)) }
	y := func(v int) int { return int(math.Round(float64(target.Min.Y) + float64(v-original.Min.Y)*sy)) }
	child := image.Rect(x(face.X*48), y(face.Y*6), x((face.X*8+face.W*8)*6), y((face.Y+face.H)*6))
	return child, target, nil
}
func (p *equipmentPrototype) shieldFaceV8(n *hdNativeRender, dst *ebiten.Image, sprites Sprites, group []portrait.Piece, body, torso, dx int, face portrait.Piece) error {
	rect, _, err := p.shieldFaceRectV8(group, body, torso, dx, face)
	if err != nil {
		return err
	}
	pic, err := pieceImage(sprites, face)
	if err != nil {
		return err
	}
	if pic.Bounds().Dx() != 40 || pic.Bounds().Dy() != 38 {
		return fmt.Errorf("DRAFT個人盾牌必須為40×38")
	}
	var op ebiten.DrawImageOptions
	op.GeoM.Scale(float64(rect.Dx())/40, float64(rect.Dy())/38)
	op.GeoM.Translate(float64(rect.Min.X), float64(rect.Min.Y))
	op.Filter = ebiten.FilterNearest
	dst.DrawImage(pic, &op)
	return nil
}
